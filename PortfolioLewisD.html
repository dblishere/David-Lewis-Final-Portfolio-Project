# Filter the dataset for the years 2001–2020
filtered_data = combined[(combined["Year"] >= 2001) & (combined["Year"] <= 2020)]

# Set up the figure
fig, ax1 = plt.subplots(figsize=(12, 6))

# Plot the GDP on the primary y-axis
ax1.plot(filtered_data["Year"], filtered_data["GDP"], label="GDP (current US$)", color="blue", marker="o")
ax1.set_xlabel("Year", fontsize=12)
ax1.set_ylabel("GDP (current US$)", fontsize=12, color="blue")
ax1.tick_params(axis="y", labelcolor="blue")
ax1.grid(alpha=0.5, linestyle="--")

# Create a secondary y-axis for Primary Enrollment
ax2 = ax1.twinx()
ax2.plot(filtered_data["Year"], filtered_data["Primary Enrollment"], label="Primary Enrollment", color="green", marker="o")
ax2.set_ylabel("Primary Enrollment", fontsize=12, color="green")
ax2.tick_params(axis="y", labelcolor="green")

# Add a title
plt.title("Overlay Line Graph of GDP and Primary Enrollment (2001–2020)", fontsize=16)
fig.tight_layout()  # Adjust layout to prevent overlapping

# Save the plot:
